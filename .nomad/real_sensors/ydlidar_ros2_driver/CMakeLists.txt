cmake_minimum_required(VERSION 3.10)
project(ydlidar_ros2_driver C CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_srvs REQUIRED)
if(DEFINED G2_SDK_ROOT)
  set(YDLIDAR_SDK_INCLUDE_DIRS "${G2_SDK_ROOT}/include")
  set(YDLIDAR_SDK_LIBRARIES "${G2_SDK_ROOT}/lib/libydlidar_sdk.a" pthread rt)
else()
  find_package(ydlidar_sdk REQUIRED)
endif()
add_executable(${PROJECT_NAME}_node src/${PROJECT_NAME}_node.cpp)
target_include_directories(${PROJECT_NAME}_node PRIVATE ${YDLIDAR_SDK_INCLUDE_DIRS})
ament_target_dependencies(${PROJECT_NAME}_node rclcpp sensor_msgs std_srvs)
target_link_libraries(${PROJECT_NAME}_node ${YDLIDAR_SDK_LIBRARIES})
install(TARGETS ${PROJECT_NAME}_node DESTINATION lib/${PROJECT_NAME})
ament_package()
