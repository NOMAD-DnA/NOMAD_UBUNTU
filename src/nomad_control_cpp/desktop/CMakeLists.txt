cmake_minimum_required(VERSION 3.16)
project(nomad_desktop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
add_executable(nomad_desktop main.cpp)
target_link_libraries(nomad_desktop PRIVATE Qt6::Widgets Qt6::Network)
if(MSVC)
  target_compile_options(nomad_desktop PRIVATE /utf-8 /W4)
else()
  target_compile_options(nomad_desktop PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS nomad_desktop RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  add_executable(nomad_desktop_tests test_console.cpp)
  target_link_libraries(nomad_desktop_tests PRIVATE Qt6::Widgets Qt6::Network Qt6::Test)
  add_test(NAME desktop_http_input COMMAND nomad_desktop_tests -platform offscreen)
endif()
