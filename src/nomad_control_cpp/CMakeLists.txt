# ROS가 없는 PC에서는 계산·상태 머신만, Humble 환경에서는 실제 ROS 노드도 빌드합니다.
cmake_minimum_required(VERSION 3.16)
project(nomad_control_cpp LANGUAGES C CXX)

find_package(ament_cmake QUIET)
option(NOMAD_BUILD_ROS2 "Build the ROS 2 VESC trial node" ${ament_cmake_FOUND})

add_library(nomad_mapping src/vesc_mapping.cpp)
target_include_directories(nomad_mapping PUBLIC include)
target_compile_features(nomad_mapping PUBLIC cxx_std_17)
set_target_properties(nomad_mapping PROPERTIES CXX_EXTENSIONS OFF)

add_library(nomad_trial src/trial_controller.cpp)
target_link_libraries(nomad_trial PUBLIC nomad_mapping)

add_library(nomad_raw_trial src/raw_trial.cpp)
target_link_libraries(nomad_raw_trial PUBLIC nomad_trial)

add_executable(nomad_convert src/main.cpp)
target_include_directories(nomad_convert PRIVATE config third_party)
target_link_libraries(nomad_convert PRIVATE nomad_mapping)

find_package(Threads REQUIRED)
add_library(nomad_gui_http INTERFACE)
target_include_directories(nomad_gui_http INTERFACE include third_party)
target_compile_features(nomad_gui_http INTERFACE cxx_std_17)
target_link_libraries(nomad_gui_http INTERFACE Threads::Threads)
if(WIN32)
  target_link_libraries(nomad_gui_http INTERFACE ws2_32 crypt32)
endif()
add_executable(nomad_gui_preview src/gui_preview.cpp)
target_link_libraries(nomad_gui_preview PRIVATE nomad_gui_http)

# 라이브러리 계약 검사와 명령행 입력 검사를 등록합니다.

if(NOMAD_BUILD_ROS2)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(rcl_interfaces REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(std_srvs REQUIRED)
  find_package(vesc_msgs REQUIRED)
  option(NOMAD_ENABLE_PLANNING "Enable typed NOMAD planning interface" ON)
  if(NOMAD_ENABLE_PLANNING)
    find_package(nomad_interfaces REQUIRED)
  endif()
  find_package(builtin_interfaces REQUIRED)
  find_package(rosidl_default_generators REQUIRED)
  find_package(ament_index_cpp REQUIRED)
  rosidl_generate_interfaces(${PROJECT_NAME} "srv/ManualControl.srv" "srv/CommissioningControl.srv" "srv/SetSteeringRange.srv" DEPENDENCIES builtin_interfaces)
  rosidl_get_typesupport_target(nomad_typesupport ${PROJECT_NAME} "rosidl_typesupport_cpp")
  ament_export_dependencies(rosidl_default_runtime)

  # The node implementation is compiled once and linked by the executable and the ROS tests,
  # so the tests exercise the same optimized code that runs on the Pi.
  add_library(nomad_controller_core STATIC src/controller_node.cpp)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(nomad_controller_core PRIVATE $<$<CONFIG:Release>:-O1>)
  endif()
  target_include_directories(nomad_controller_core PUBLIC include PRIVATE config third_party)
  target_link_libraries(nomad_controller_core nomad_raw_trial "${nomad_typesupport}")
  ament_target_dependencies(nomad_controller_core rclcpp rcl_interfaces std_msgs std_srvs vesc_msgs sensor_msgs)
  if(NOMAD_ENABLE_PLANNING)
    target_compile_definitions(nomad_controller_core PRIVATE NOMAD_ENABLE_PLANNING)
    ament_target_dependencies(nomad_controller_core nomad_interfaces)
  endif()

  add_executable(controller_node src/controller_main.cpp)
  target_link_libraries(controller_node nomad_controller_core)
  ament_target_dependencies(controller_node rclcpp)

  add_executable(gui_server src/gui_server.cpp)
  target_link_libraries(gui_server nomad_gui_http "${nomad_typesupport}")
  ament_target_dependencies(gui_server rclcpp std_msgs std_srvs ament_index_cpp)

  if(BUILD_TESTING)
    add_executable(controller_ros_tests test/test_controller_ros.cpp)
    # 실제 노드+HTTP 구현을 포함한 테스트는 Pi에서 과도한 최적화 컴파일 비용을 피합니다.
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      target_compile_options(controller_ros_tests PRIVATE -O0)
    endif()
    target_include_directories(controller_ros_tests PRIVATE third_party)
    target_link_libraries(controller_ros_tests nomad_controller_core nomad_gui_http)
    ament_target_dependencies(controller_ros_tests rclcpp rcl_interfaces std_msgs std_srvs vesc_msgs ament_index_cpp sensor_msgs)
    if(NOMAD_ENABLE_PLANNING)
      target_compile_definitions(controller_ros_tests PRIVATE NOMAD_ENABLE_PLANNING)
      ament_target_dependencies(controller_ros_tests nomad_interfaces)
    endif()
    add_test(NAME controller_ros_integration COMMAND controller_ros_tests)
    # The full suite runs about 62 s on a desktop PC; leave margin for slower machines.
    set_tests_properties(controller_ros_integration PROPERTIES TIMEOUT 150)
    if(NOMAD_ENABLE_PLANNING)
      # Device-free contract: launch the already built production node once.
      add_executable(test_module_pause_contract test/test_module_pause_contract.cpp)
      target_compile_features(test_module_pause_contract PRIVATE cxx_std_17)
      if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(test_module_pause_contract PRIVATE -O0)
      endif()
      ament_target_dependencies(test_module_pause_contract rclcpp std_msgs std_srvs vesc_msgs nomad_interfaces)
      add_dependencies(test_module_pause_contract controller_node)
      add_test(NAME module_pause_auto_resume
        COMMAND test_module_pause_contract $<TARGET_FILE:controller_node>)
      set_tests_properties(module_pause_auto_resume PROPERTIES TIMEOUT 35
        ENVIRONMENT "ROS_DOMAIN_ID=229;ROS_LOCALHOST_ONLY=1;RMW_IMPLEMENTATION=rmw_fastrtps_cpp")
      add_executable(rpm_floor_ros_tests test/test_rpm_floor_ros.cpp)
      if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(rpm_floor_ros_tests PRIVATE -O0)
      endif()
      target_include_directories(rpm_floor_ros_tests PRIVATE third_party)
      target_link_libraries(rpm_floor_ros_tests nomad_controller_core)
      target_compile_definitions(rpm_floor_ros_tests PRIVATE NOMAD_ENABLE_PLANNING)
      ament_target_dependencies(rpm_floor_ros_tests rclcpp rcl_interfaces std_msgs std_srvs vesc_msgs sensor_msgs nomad_interfaces)
      add_test(NAME rpm_running_floor_ros COMMAND rpm_floor_ros_tests)
      set_tests_properties(rpm_running_floor_ros PROPERTIES TIMEOUT 30
        ENVIRONMENT "ROS_DOMAIN_ID=232;ROS_LOCALHOST_ONLY=1;RMW_IMPLEMENTATION=rmw_fastrtps_cpp")
    endif()
  endif()
  install(TARGETS controller_node gui_server nomad_convert DESTINATION lib/${PROJECT_NAME})
  install(TARGETS nomad_mapping nomad_trial nomad_raw_trial
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
    RUNTIME DESTINATION bin)
  install(DIRECTORY config web scripts launch DESTINATION share/${PROJECT_NAME})
  install(FILES third_party/httplib.LICENSE third_party/json.LICENSE third_party/SOURCES.txt DESTINATION share/${PROJECT_NAME}/third_party)
  install(FILES GUIDE.ko.txt ROS_TRIAL_GUIDE.ko.txt GUI_GUIDE.ko.txt GUI_START.ko.txt RAW_COMMISSION_GUIDE.ko.txt STABILITY_GUIDE.ko.txt STEERING_CALIBRATION.ko.txt DESTINATION share/${PROJECT_NAME})
  install(FILES AUTONOMY_GUIDE.ko.txt DESTINATION share/${PROJECT_NAME})
endif()

foreach(target nomad_raw_trial raw_trial_tests nomad_mapping nomad_trial nomad_convert mapping_tests trial_tests nomad_controller_core controller_node controller_ros_tests gui_server nomad_gui_preview gui_http_tests)
  if(TARGET ${target})
    if(MSVC)
      target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
      target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
  endif()
endforeach()

if(NOMAD_BUILD_ROS2)
  ament_package()
endif()
