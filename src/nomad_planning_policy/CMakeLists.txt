cmake_minimum_required(VERSION 3.16)
project(nomad_planning_policy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(nomad_planning_core REQUIRED)
include(CTest)
add_library(nomad_planning_policy SHARED src/policy.cpp)
target_include_directories(nomad_planning_policy PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(nomad_planning_policy nomad_planning_core::nomad_planning_core)
target_compile_options(nomad_planning_policy PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS nomad_planning_policy EXPORT export_nomad_planning_policy ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_nomad_planning_policy HAS_LIBRARY_TARGET)
ament_export_dependencies(nomad_planning_core)
