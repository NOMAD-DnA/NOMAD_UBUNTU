cmake_minimum_required(VERSION 3.16)
project(nomad_planning_cpp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NOMAD_BUILD_ROS "Build the ROS2 planning node" ON)
include(CTest)
find_package(Threads REQUIRED)
# 2026-11-13 split: core (cost/global search) and policy (local corridor/RPP/navigator) are separate packages.
find_package(nomad_planning_policy REQUIRED)
if(NOMAD_BUILD_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(std_msgs REQUIRED)
  find_package(std_srvs REQUIRED)
  find_package(nomad_interfaces REQUIRED)
  add_executable(planner_node src/planner_node.cpp)
  target_include_directories(planner_node PRIVATE include)
  target_link_libraries(planner_node nomad_planning_policy::nomad_planning_policy Threads::Threads)
  ament_target_dependencies(planner_node rclcpp nav_msgs geometry_msgs std_msgs std_srvs nomad_interfaces)
  target_compile_options(planner_node PRIVATE -Wall -Wextra -Wpedantic)
  if(BUILD_TESTING)
    add_executable(ros_contract_test tests/ros_contract_test.cpp)
    ament_target_dependencies(ros_contract_test rclcpp nav_msgs geometry_msgs std_srvs nomad_interfaces)
    install(TARGETS ros_contract_test RUNTIME DESTINATION lib/${PROJECT_NAME})
  endif()
  install(TARGETS planner_node ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY include/ DESTINATION include)
  install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
  install(PROGRAMS scripts/run_planning.sh scripts/run_planning_output.sh scripts/stop_planning.sh DESTINATION lib/${PROJECT_NAME})
  ament_export_include_directories(include)
  ament_export_dependencies(nomad_planning_policy)
  ament_package()
endif()
if(BUILD_TESTING)
  add_executable(trajectory_writer_test tests/test_trajectory_writer.cpp)
  target_include_directories(trajectory_writer_test PRIVATE include)
  target_link_libraries(trajectory_writer_test Threads::Threads)
  target_compile_options(trajectory_writer_test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME trajectory_writer_test COMMAND trajectory_writer_test)
  set_tests_properties(trajectory_writer_test PROPERTIES TIMEOUT 10)
  add_executable(status_diagnostics_test tests/test_status_diagnostics.cpp)
  target_include_directories(status_diagnostics_test PRIVATE include)
  target_compile_options(status_diagnostics_test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME status_diagnostics_test COMMAND status_diagnostics_test)
  add_executable(lpp_diagnostics_test tests/test_lpp_diagnostics.cpp)
  target_include_directories(lpp_diagnostics_test PRIVATE include)
  target_compile_options(lpp_diagnostics_test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME lpp_diagnostics_test COMMAND lpp_diagnostics_test)
  add_executable(latest_json_writer_test tests/test_latest_json_writer.cpp)
  target_include_directories(latest_json_writer_test PRIVATE include)
  target_link_libraries(latest_json_writer_test Threads::Threads)
  target_compile_options(latest_json_writer_test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME latest_json_writer_test COMMAND latest_json_writer_test)
  set_tests_properties(latest_json_writer_test PROPERTIES TIMEOUT 10)
endif()
