cmake_minimum_required(VERSION 3.16)
project(nomad_planning_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
include(CTest)
# 2026-11-13 split of nomad_planning_cpp: same sources and flags; headers keep the nomad_planning_cpp/ include path.
add_library(nomad_planning_core SHARED src/core.cpp)
target_include_directories(nomad_planning_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_options(nomad_planning_core PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS nomad_planning_core EXPORT export_nomad_planning_core ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_nomad_planning_core HAS_LIBRARY_TARGET)
if(BUILD_TESTING)
  add_executable(core_test tests/test_core.cpp)
  target_link_libraries(core_test nomad_planning_core)
  add_test(NAME core_test COMMAND core_test)
  add_executable(nomad_planning_parity_cli tests/parity_cli.cpp)
  target_link_libraries(nomad_planning_parity_cli nomad_planning_core)
endif()
ament_package()
