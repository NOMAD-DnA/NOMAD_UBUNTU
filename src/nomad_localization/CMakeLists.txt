cmake_minimum_required(VERSION 3.16)
project(nomad_localization)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(Ceres REQUIRED)
find_package(Eigen3 REQUIRED)
foreach(dep rclcpp nav_msgs sensor_msgs geometry_msgs std_msgs std_srvs diagnostic_msgs nomad_interfaces)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(fusion_node src/fusion_node.cpp)
target_link_libraries(fusion_node Ceres::ceres Eigen3::Eigen)
ament_target_dependencies(fusion_node rclcpp nav_msgs sensor_msgs geometry_msgs std_msgs std_srvs diagnostic_msgs nomad_interfaces)
target_compile_options(fusion_node PRIVATE -Wall -Wextra)
install(TARGETS fusion_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/vio_supervisor.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
ament_package()
