project(nomad_semantic_adapter LANGUAGES CXX)

set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(rclcpp REQUIRED)

foreach(package sensor_msgs nav_msgs std_msgs diagnostic_msgs nomad_interfaces)
  find_package(${package} REQUIRED)
endforeach()
# Reuse the audited native-only imported ROS target closure of the frozen source.
include(${CMAKE_CURRENT_SOURCE_DIR}/../source/cmake/native_ros_targets.cmake)
function(adapter_target target source)
  add_executable(${target} ${source})
  target_include_directories(${target} PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../source/vendor/include)
  target_link_libraries(${target} PRIVATE ${NOMAD_ROS_TARGETS})
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endfunction()
adapter_target(nomad_semantic_id_adapter semantic_id_adapter.cpp)
