cmake_minimum_required(VERSION 3.16)
project(nomad_g2b_native_driver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(nomad_interfaces REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(visualization_msgs REQUIRED)
include(${CMAKE_CURRENT_LIST_DIR}/../native_ros_targets.cmake)
foreach(package geometry_msgs std_srvs visualization_msgs)
  foreach(kind rosidl_generator_cpp rosidl_typesupport_cpp)
    set(target "${package}::${package}__${kind}")
    nomad_native_closure("${target}")
    list(APPEND NOMAD_ROS_TARGETS "${target}")
  endforeach()
endforeach()
if(NOT EXISTS "${NOMAD_LIDAR_SOURCE}/src/ydlidar_ros2_driver_node.cpp" OR NOT EXISTS "${G2_SDK_ROOT}/lib/libydlidar_sdk.a")
  message(FATAL_ERROR "Verified LiDAR source or built SDK is missing")
endif()
add_executable(ydlidar_ros2_driver_node "${NOMAD_LIDAR_SOURCE}/src/ydlidar_ros2_driver_node.cpp")
target_include_directories(ydlidar_ros2_driver_node PRIVATE "${G2_SDK_ROOT}/include" "${NOMAD_LIDAR_SOURCE}/src")
target_link_libraries(ydlidar_ros2_driver_node PRIVATE "${G2_SDK_ROOT}/lib/libydlidar_sdk.a" pthread rt ${NOMAD_ROS_TARGETS})
install(TARGETS ydlidar_ros2_driver_node DESTINATION lib/ydlidar_ros2_driver)
