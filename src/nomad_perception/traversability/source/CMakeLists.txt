cmake_minimum_required(VERSION 3.16)
project(nomad_costmap_cpp VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-fno-fast-math -ffp-contract=off)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)
add_library(nomad_json INTERFACE)
target_include_directories(nomad_json INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/vendor/include)
add_library(nlohmann_json::nlohmann_json ALIAS nomad_json)
find_package(Threads REQUIRED)
option(NOMAD_WITH_ROS "Build the ROS 2 runtime" ON)
option(NOMAD_SANITIZE "Enable address and undefined behavior checks" OFF)
add_library(nomad_costmap_core src/core.cpp src/camera_model.cpp)
target_include_directories(nomad_costmap_core PUBLIC include)
target_link_libraries(nomad_costmap_core PUBLIC Eigen3::Eigen nlohmann_json::nlohmann_json Threads::Threads)
target_compile_options(nomad_costmap_core PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
if(NOMAD_SANITIZE)
  target_compile_options(nomad_costmap_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(nomad_costmap_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(nomad_core_probe src/core_probe.cpp)
target_link_libraries(nomad_core_probe PRIVATE nomad_costmap_core)
enable_testing()
if(NOMAD_WITH_ROS)
  include(cmake/ros.cmake)
endif()
install(TARGETS nomad_costmap_core nomad_core_probe DESTINATION lib/nomad_costmap_cpp)
install(FILES config/semantic.real.json config/slope.json config/g2b.yaml config/classes.csv
  config/oak_g2b_selected_20261030.json config/zed2_g2b_selected_20261030.json
  DESTINATION share/nomad_costmap_cpp/config)
