cmake_minimum_required(VERSION 3.16)
project(nomad_indoor_mapper LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-fno-fast-math -ffp-contract=off -Wall -Wextra)
find_package(Eigen3 REQUIRED)
find_package(nlohmann_json QUIET)
if(NOT nlohmann_json_FOUND)
  add_library(indoor_json_headers INTERFACE)
  target_include_directories(indoor_json_headers INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/vendor/include)
  add_library(nlohmann_json::nlohmann_json ALIAS indoor_json_headers)
endif()
add_library(indoor_core src/indoor.cpp src/ray_trace.cpp src/legacy_geometry.cpp src/legacy_cost.cpp)
target_include_directories(indoor_core PUBLIC include)
target_link_libraries(indoor_core PUBLIC Eigen3::Eigen nlohmann_json::nlohmann_json)
add_executable(indoor_probe src/probe.cpp)
target_link_libraries(indoor_probe indoor_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(indoor_rules_test tests/indoor_rules_test.cpp)
  target_link_libraries(indoor_rules_test indoor_core)
  foreach(group geometry updates retention poses config)
    add_test(NAME indoor_${group} COMMAND indoor_rules_test ${group} ${CMAKE_CURRENT_SOURCE_DIR}/config/indoor.json)
  endforeach()
endif()
option(INDOOR_WITH_ROS "Build the ROS 2 mapper" ON)
if(INDOOR_WITH_ROS)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(diagnostic_msgs REQUIRED)
  find_package(tf2_ros REQUIRED)
  find_package(nomad_interfaces REQUIRED)
  add_executable(nomad_indoor_mapper src/mapper.cpp)
  target_link_libraries(nomad_indoor_mapper indoor_core)
  ament_target_dependencies(nomad_indoor_mapper rclcpp sensor_msgs nav_msgs diagnostic_msgs tf2_ros nomad_interfaces)
  install(TARGETS nomad_indoor_mapper indoor_probe indoor_core DESTINATION lib/${PROJECT_NAME})
  install(DIRECTORY config scripts DESTINATION share/${PROJECT_NAME})
  ament_package()
endif()
