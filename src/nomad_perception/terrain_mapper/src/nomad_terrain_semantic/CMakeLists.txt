cmake_minimum_required(VERSION 3.16)
project(nomad_terrain_semantic LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-fno-fast-math -ffp-contract=off)
find_package(ament_cmake REQUIRED)
find_package(nomad_terrain_core REQUIRED)
add_library(nomad_terrain_semantic SHARED src/semantic.cpp)
target_link_libraries(nomad_terrain_semantic PUBLIC nomad_terrain_core::nomad_terrain_core)
target_compile_options(nomad_terrain_semantic PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS nomad_terrain_semantic EXPORT nomad_terrain_semanticTargets
  LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
ament_export_targets(nomad_terrain_semanticTargets HAS_LIBRARY_TARGET)
ament_export_dependencies(nomad_terrain_core)
ament_package()
