cmake_minimum_required(VERSION 3.16)
project(nomad_terrain_mapper LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-fno-fast-math -ffp-contract=off)
find_package(ament_cmake REQUIRED)
find_package(nomad_terrain_core REQUIRED)
find_package(nomad_terrain_semantic REQUIRED)
find_package(nomad_terrain_cost REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(nomad_interfaces REQUIRED)
add_executable(nomad_terrain_mapper src/mapper.cpp)
target_include_directories(nomad_terrain_mapper PRIVATE include)
target_link_libraries(nomad_terrain_mapper nomad_terrain_core::nomad_terrain_core
  nomad_terrain_semantic::nomad_terrain_semantic nomad_terrain_cost::nomad_terrain_cost)
target_compile_options(nomad_terrain_mapper PRIVATE -Wall -Wextra)
ament_target_dependencies(nomad_terrain_mapper rclcpp sensor_msgs nav_msgs std_msgs diagnostic_msgs tf2_ros nomad_interfaces)
install(TARGETS nomad_terrain_mapper DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config scripts DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(oak_horizontal_crop_test src/oak_horizontal_crop_test.cpp)
  target_include_directories(oak_horizontal_crop_test PRIVATE include)
  target_link_libraries(oak_horizontal_crop_test nomad_terrain_core::nomad_terrain_core)
  add_test(NAME oak_horizontal_crop_test COMMAND oak_horizontal_crop_test)
  add_executable(oak_height_override_test src/oak_height_override_test.cpp)
  target_include_directories(oak_height_override_test PRIVATE include)
  target_link_libraries(oak_height_override_test nomad_terrain_core::nomad_terrain_core)
  add_test(NAME oak_height_override_test COMMAND oak_height_override_test)
endif()
ament_package()
