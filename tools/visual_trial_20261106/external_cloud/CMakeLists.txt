cmake_minimum_required(VERSION 3.16)
project(nomad_external_cloud LANGUAGES CXX)

option(BUILD_BRIDGE "Build the ROS stereo cloud bridge" ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d)
find_package(Threads REQUIRED)

add_executable(cloud_geometry_check src/cloud_geometry_check.cpp)
target_compile_features(cloud_geometry_check PRIVATE cxx_std_17)
target_include_directories(cloud_geometry_check PRIVATE include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(cloud_geometry_check PRIVATE ${OpenCV_LIBS})
target_compile_options(cloud_geometry_check PRIVATE -Wall -Wextra -Wpedantic)

if(BUILD_BRIDGE)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(std_msgs REQUIRED)

  add_executable(external_stereo_cloud src/external_stereo_cloud.cpp)
  target_compile_features(external_stereo_cloud PRIVATE cxx_std_17)
  target_include_directories(external_stereo_cloud PRIVATE include ${OpenCV_INCLUDE_DIRS})
  target_compile_options(external_stereo_cloud PRIVATE -Wall -Wextra -Wpedantic)
  # Keep this independent executable free of message-generator Python linkage.
  function(cpp_only target)
    if(NOT TARGET "${target}")
      return()
    endif()
    get_property(visited GLOBAL PROPERTY EXTERNAL_CLOUD_VISITED)
    if("${target}" IN_LIST visited)
      return()
    endif()
    set_property(GLOBAL APPEND PROPERTY EXTERNAL_CLOUD_VISITED "${target}")
    get_target_property(dependencies "${target}" INTERFACE_LINK_LIBRARIES)
    if(NOT dependencies)
      return()
    endif()
    set(clean "")
    foreach(dependency IN LISTS dependencies)
      if(dependency MATCHES "__rosidl_generator_py")
        continue()
      endif()
      list(APPEND clean "${dependency}")
      if(TARGET "${dependency}")
        cpp_only("${dependency}")
      endif()
    endforeach()
    set_property(TARGET "${target}" PROPERTY INTERFACE_LINK_LIBRARIES "${clean}")
  endfunction()
  set(ros_dependencies rclcpp::rclcpp
      sensor_msgs::sensor_msgs__rosidl_typesupport_cpp
      std_msgs::std_msgs__rosidl_typesupport_cpp)
  foreach(dependency IN LISTS ros_dependencies)
    cpp_only("${dependency}")
  endforeach()
  target_link_libraries(external_stereo_cloud PRIVATE ${ros_dependencies} ${OpenCV_LIBS} Threads::Threads)
  install(TARGETS external_stereo_cloud cloud_geometry_check DESTINATION lib/${PROJECT_NAME})
  ament_package()
endif()
