# Native Humble's imported message interfaces can include Python generators.
# This application uses only C++ typesupport; do not alter the installed ROS.
function(nomad_cpp_only_interface target)
  if(NOT TARGET "${target}")
    return()
  endif()
  get_property(visited GLOBAL PROPERTY NOMAD_CPP_ONLY_VISITED)
  if("${target}" IN_LIST visited)
    return()
  endif()
  set_property(GLOBAL APPEND PROPERTY NOMAD_CPP_ONLY_VISITED "${target}")
  get_target_property(dependencies "${target}" INTERFACE_LINK_LIBRARIES)
  if(NOT dependencies)
    return()
  endif()
  set(filtered "")
  foreach(dependency IN LISTS dependencies)
    if(dependency MATCHES "__rosidl_generator_py")
      continue()
    endif()
    list(APPEND filtered "${dependency}")
    if(TARGET "${dependency}")
      nomad_cpp_only_interface("${dependency}")
    endif()
  endforeach()
  set_property(TARGET "${target}" PROPERTY INTERFACE_LINK_LIBRARIES "${filtered}")
endfunction()
set(ros_cpp_targets rclcpp::rclcpp tf2_ros::tf2_ros)
foreach(package nav_msgs sensor_msgs geometry_msgs std_msgs std_srvs diagnostic_msgs nomad_interfaces tf2_msgs)
  list(APPEND ros_cpp_targets "${package}::${package}__rosidl_typesupport_cpp")
endforeach()
foreach(target IN LISTS ros_cpp_targets)
  nomad_cpp_only_interface("${target}")
endforeach()
